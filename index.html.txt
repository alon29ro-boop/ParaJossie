/* =========================

function crearPetalo() {

    const petalo = document.createElement("div");

    petalo.className = "fallingPetal";

    petalo.style.left =
        Math.random() * 100 + "vw";

    petalo.style.setProperty(
        "--x",
        (Math.random() * 160 - 80) + "px"
    );

    petalo.style.animationDuration =
        (5 + Math.random() * 5) + "s";

    petalo.style.transform =
        `rotate(${Math.random() * 360}deg)`;

    petalo.style.width =
        (9 + Math.random() * 9) + "px";

    petalo.style.height =
        (15 + Math.random() * 12) + "px";

    document.body.appendChild(petalo);

    setTimeout(() => {
        petalo.remove();
    }, 11000);
}


/* Primera lluvia */

setTimeout(() => {

    for (let i = 0; i < 20; i++) {

        setTimeout(() => {
            crearPetalo();
        }, i * 180);

    }

}, 5000);


/* Lluvia continua */

const lluvia = setInterval(() => {

    crearPetalo();

}, 850);


/* =========================
   MOSTRAR CARTA
========================= */

    clearInterval(lluvia);

    document.getElementById("scene")

    document.getElementById("hint")

    setTimeout(() => {

        document.getElementById("card")
            .classList.add("show");

    }, 1100);

}, 8000);
